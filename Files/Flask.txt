def show_products():
    # ❌ الخطأ 1: Undefined Error - products غير معرف (إذا تم حذفه)
    for p in products:
        p["image_url"] = url_for('static', filename=f"images/{p['image']}")
    return render_template("products.html", products=products)

@app.route("/product/<int:pid>")
def product_details(pid):
    # ❌ الخطأ 27: Index Out of Range - pid قد يكون خارج النطاق
    product = products[pid]
    if product:
        product["image_url"] = url_for('static', filename=f"images/{product['image']}")
    return render_template("products.html", product=product)

@app.route("/add", methods=["GET", "POST"])
def add_product():
    if request.method == "POST":
        name = request.form["name"]
        price = float(request.form["price"])
        file = request.files["image"]
        filename = secure_filename(file.filename)
        file.save(os.path.join(app.config['UPLOAD_FOLDER'], filename))

        new_id = len(products) + 1
        products.append({
            "id": new_id,
            "name": name,
            "price": price,
            "image": filename,
            "extra": sample_product["price"]  # ❌ Key Error
        })
        return redirect("/")
    return render_template("products.html", add_form=True)

@app.route("/delete/<int:pid>", methods=["POST"])
def delete_product(pid):
    global products
    products = [p for p in products if p["id"] != pid]
    return redirect("/")
